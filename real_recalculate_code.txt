def load_history():
    history = []

    for raw in RAW_DRAWS:
        parsed = list(parse_draw(raw))

        if len(parsed) != 7:
            raise ValueError(
                f"Ожидалось 7 значений, получено "
                f"{len(parsed)}: {parsed}"
            )

        main = parsed[:5]
        euro = parsed[5:7]
        history.append(
            (
                list(main),
                list(euro),
            )
        )

    if len(history) != 81:
        raise RuntimeError(
            f"Ожидалось 81 состояние, получено {len(history)}"
        )

    return history


def build_history_stats(history):
    main_frequency = Counter()
    euro_frequency = Counter()

    main_last = {
        number: -1
        for number in range(1, 51)
    }

    euro_last = {
        number: -1
        for number in range(1, 13)
    }

    main_euro_coupling = defaultdict(Counter)

    for index, (main, euro) in enumerate(history):
        main_frequency.update(main)
        euro_frequency.update(euro)

        for number in main:
            main_last[number] = index

            for euro_number in euro:
                main_euro_coupling[number][euro_number] += 1

        for euro_number in euro:
            euro_last[euro_number] = index

    return {
        "main_frequency": main_frequency,
        "euro_frequency": euro_frequency,
        "main_last": main_last,
        "euro_last": euro_last,
        "coupling": main_euro_coupling,
        "history_size": len(history),
    }


def toroidal_point(
    number,
    limit,
    zone,
    reflected=False,
):
    """
    Ограниченная тороидальная труба.

    Хиральная ветвь меняет ориентацию phi,
    но не создаёт числового взрыва и не использует
    искусственное ограничение state=1000.
    """

    phi = (
        2.0
        * math.pi
        * (number - 1)
        / limit
    )

    if reflected:
        phi = -phi

    zone_order = {
        "ROOT": 0,
        "JILS": 1,
        "TAIL": 2,
    }

    theta = (
        2.0
        * math.pi
        * (zone_order[zone] + 0.5)
        / 3.0
    )

    major_radius = 2.0
    minor_radius = ZONE_RADIUS[zone]

    return (
        (
            major_radius
            + minor_radius * math.cos(theta)
        )
        * math.cos(phi),

        (
            major_radius
            + minor_radius * math.cos(theta)
        )
        * math.sin(phi),

        minor_radius * math.sin(theta),
    )

def main_field(
    vector,
    zone,
    stats,
    reflected=False,
):
    frequency = stats["main_frequency"]
    maximum_frequency = max(
        frequency.values()
    )

    anchors = []

    for position, number in enumerate(vector):
        anchor_zone = ZONE_INDEX[position]

        anchors.append(
            (
                toroidal_point(
                    number,
                    50,
                    anchor_zone,
                    reflected=reflected,
                ),
                anchor_zone,
            )
        )

    scores = {}

    for candidate in range(1, 51):
        candidate_point = toroidal_point(
            candidate,
            50,
            zone,
            reflected=False,
        )

        geometry_score = 0.0

        for anchor_point, anchor_zone in anchors:
            zone_factor = (
                1.0
                if anchor_zone == zone
                else 0.38
            )

            geometry_score += (
                zone_factor
                * math.exp(
                    -distance(
                        candidate_point,
                        anchor_point,
                    )
                    / 1.10
                )
            )

        frequency_score = (
            frequency[candidate]
            / maximum_frequency
        )

        gap_score = (
            stats["history_size"]
            - 1
            - stats["main_last"][candidate]
        ) / stats["history_size"]

        scores[candidate] = (
            0.68 * geometry_score
            + 0.20 * frequency_score
            + 0.12 * gap_score
        )

    return scores


def select_unique(
    scores,
    count,
    used,
):
    selected = []

    order = sorted(
        scores,
        key=lambda number: (
            -scores[number],
            number,
        ),
    )

    for number in order:
        if number in used:
            continue

        selected.append(number)
        used.add(number)

        if len(selected) == count:
            break

    return selected

def euro_field(
    main_vector,
    stats,
    expert_index,
    reflected=False,
):
    frequency = stats["euro_frequency"]
    maximum_frequency = max(
        frequency.values()
    )

    maximum_coupling = max(
        1,
        max(
            sum(
                stats["coupling"][main_number][euro]
                for main_number in main_vector
            )
            for euro in range(1, 13)
        ),
    )

    scores = {}

    for euro in range(1, 13):
        frequency_score = (
            frequency[euro]
            / maximum_frequency
        )

        gap_score = (
            stats["history_size"]
            - 1
            - stats["euro_last"][euro]
        ) / stats["history_size"]

        coupling_score = (
            sum(
                stats["coupling"][main_number][euro]
                for main_number in main_vector
            )
            / maximum_coupling
        )

        phase = (
            2.0
            * math.pi
            * (euro - 1)
            / 12.0
        )

        expert_phase = (
            expert_index
            * math.pi
            / 9.0
        )

        if reflected:
            orientation = math.cos(
                -phase + expert_phase
            )
        else:
            orientation = math.cos(
                phase + expert_phase
            )

        orientation_score = (
            orientation + 1.0
        ) / 2.0

        scores[euro] = (
            0.38 * frequency_score
            + 0.36 * coupling_score
            + 0.16 * gap_score
            + 0.10 * orientation_score
        )

    return scores

def average_selected_score(
    scores,
    numbers,
):
    if not numbers:
        return 0.0

    return sum(
        scores[number]
        for number in numbers
    ) / len(numbers)


def update_zones(
    storage,
    vector,
    weight=1.0,
):
    for zone, positions in ZONE_POSITIONS.items():
        for position in positions:
            storage[zone][vector[position]] += weight
